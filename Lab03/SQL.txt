create database LibraryDB;
\c LibraryDB;

create table Authors (
    author_id serial primary key,
    name varchar(255) not null,
    birth_date date,
    country varchar(100)
);

create table Books (
    book_id serial primary key,
    title varchar(255) not null,
    author_id int references Authors(author_id) on delete cascade,
    isbn varchar(20),
    published_year int,
);

create table Borrowers (
    borrower_id serial primary key,
    name varchar(255) not null,
    email varchar(255) unique not null,
    phone varchar(15) unique
);

create table Loans (
    loan_id serial primary key,
    book_id int references Books(book_id) on delete cascade,
    borrower_id int references Borrowers(borrower_id) on delete cascade,
    loan_date DATE not null,
    return_date DATE not null,
    returned boolean default false
);

create view PopularBooks as
select Books.book_id, Books.title, COUNT(Loans.loan_id) as total_loans
from Books
join Loans on Books.book_id = Loans.book_id
group by Books.book_id, Books.title
order by total_loans desc;
select * from PopularBooks;

create or replace function BorrowBook(p_book_id INT, p_borrower_id INT, p_return_date DATE)
retrurns void as $$
begin
    insert into Loans (book_id, borrower_id, loan_date, return_date, returned)
    values (p_book_id, p_borrower_id, CURDATE(), p_return_date, false);
end;
$$ language plpgsql;
